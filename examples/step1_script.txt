# Step 1 play scripts: super bullet throw, screen pass, stop-and-go, light injuries.
# Needs mod build v89 or later. Paste into the Workshop's Script tab (below the
# Hail Mary script if you use it) and Save to Files as rbmod_script.txt.
# To tune a feature, change the numbers in its first "on presnap" block.
# To turn a feature off, set its  ..._on  line to 0.
# Everything this file does is logged in rbmod.log on lines starting with "s1:".

# ============================================================
# 1. SUPER BULLET THROW
# Completions fill a meter. When it's full, a "Super throw" button appears:
# tap it, throw, and the receiver nearest the landing spot runs under it
# with no interception chance.
# ============================================================

on presnap
  # --- Super throw: change these ---
  super_on = 1                # 1 = on, 0 = off
  meter_per_catch = 20        # meter gained per completion (100 = full)
  meter_per_miss = 10         # meter lost per incompletion or interception
  super_needs = 100           # meter needed for a super throw
  super_slow_defenders = 1    # 1 = the 2 defenders nearest the catch are slowed
  super_def_speed = 70        # their speed during a super throw, in percent
  if meter == nil
    meter = 0
  end
end

on presnap
  if super_on
    msg("Super meter " + str(min(100, floor(meter * 100 / super_needs))) + "%")
    if meter >= super_needs
      button("Super throw")
    end
  end
end

on catch
  if super_on and not pressed("Super throw")
    meter = min(super_needs, meter + meter_per_catch)
    log("s1: meter", meter)
  end
end

on incomplete
  if super_on and not pressed("Super throw")
    meter = max(0, meter - meter_per_miss)
    log("s1: meter", meter)
  end
end

on intercepted
  if super_on and not pressed("Super throw")
    meter = max(0, meter - meter_per_miss)
    log("s1: meter", meter)
  end
end

on snap
  if super_on and pressed("Super throw")
    tune("pick_base", 0)
    tune("pick_known", 0)
    log("s1: super throw")
  end
end

on frame
  if super_on and pressed("Super throw") and thrown()
    for r in nearest(receivers(), landing_x(), landing_y(), 1)
      steer(r, landing_x(), landing_y(), 1)
    end
    if super_slow_defenders
      for d in nearest(defenders(), landing_x(), landing_y(), 2)
        speed(d, super_def_speed)
      end
    end
  end
end

on play_end
  if super_on and pressed("Super throw")
    meter = 0
  end
end

# ============================================================
# 2. SCREEN PASS
# A "Screen" button. At the snap the back runs a flat route and the linemen
# nearest him release toward the side to block for him.
# ============================================================

on presnap
  # --- Screen pass: change these ---
  screen_on = 1          # 1 = on, 0 = off
  screen_side = 0        # 1 = top side, -1 = bottom side, 0 = the side with more room
  screen_depth = 8       # how many yards toward the sideline the blockers set up, from where the back starts
  screen_linemen = 3     # how many linemen release to block
  screen_time = 3        # seconds after the snap the blockers keep heading out
  if screen_on
    button("Screen")
  end
end

on snap
  if screen_on and pressed("Screen")
    scr_dy = 1
    if screen_side == 1
      scr_dy = -1
    elif screen_side == 0 and back().y - field_top() > field_bottom() - back().y
      scr_dy = -1
    end
    scr_y = max(field_top() + 20, min(field_bottom() - 20, back().y + scr_dy * yard(screen_depth)))
    scr_x = los_x() + fwd() * yard(1)
    scr_logged = 0
    route(back(), "route_flat")
    log("s1: screen side", scr_dy, "spot", scr_x, scr_y)
  end
end

on frame
  if screen_on and pressed("Screen") and since_snap() < screen_time
    n = 0
    for p in nearest(offense(), los_x(), back().y, screen_linemen + 3)
      ok = 1
      if p == qb() or p == back()
        ok = 0
      end
      for r in receivers()
        if p == r
          ok = 0
        end
      end
      if ok and n < screen_linemen
        steer(p, scr_x, scr_y, 0.8)
        n = n + 1
        if not scr_logged
          log("s1: screen blocker", p.x, p.y)
        end
      end
    end
    scr_logged = 1
  end
end

# ============================================================
# 3. TRICK PLAYS: STOP-AND-GO, plus a test for later trick plays
# "Stop and go" gives every receiver the game's own slant-and-go route.
# The probe only writes to rbmod.log: it records how the game keeps track of
# who has the ball, which decides whether hook-and-lateral and flea flicker
# can be scripts. Turn it off with probe_on = 0 once the log has been sent.
# ============================================================

on presnap
  # --- Trick plays: change these ---
  sng_on = 1      # 1 = on, 0 = off
  probe_on = 1    # 1 = log ball-holder info, 0 = off
  last_state = -1
  last_holder = -1
  probe_logs = 0
  if sng_on
    button("Stop and go")
  end
end

on snap
  if sng_on and pressed("Stop and go")
    for r in receivers()
      route(r, "route_slant_fake")
    end
    log("s1: stop and go")
  end
end

on frame
  if probe_on and state() != last_state
    log("s1: ball state", state(), since_snap())
    last_state = state()
  end
end

# Kept apart: if the holder isn't a readable number, only this block switches off.
on frame
  if probe_on and probe_logs < 40
    h = ball().holder
    if h != last_holder
      log("s1: holder", h, since_snap())
      last_holder = h
      probe_logs = probe_logs + 1
    end
  end
end
