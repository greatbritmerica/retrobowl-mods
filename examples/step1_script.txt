# Step 1 play scripts: super bullet throw, screen pass, stop-and-go, light injuries.
# Needs mod build v89 or later. Paste into the Workshop's Script tab (below the
# Hail Mary script if you use it) and Save to Files as rbmod_script.txt.
# To tune a feature, change the numbers in its first "on presnap" block.
# To turn a feature off, set its  ..._on  line to 0.
# Everything this file does is logged in rbmod.log on lines starting with "s1:".
# Play on EXTREME difficulty: the meter and injuries use events the mod only
# sends on Extreme. The screen pass and stop-and-go work on any difficulty.
# The game shows at most 3 buttons before a snap. When the Hail Mary button is
# up (end of a half), one of these may not appear.

# ============================================================
# 1. SUPER BULLET THROW
# Completions fill a meter. When it's full, a "Super throw" button appears.
# Tap it, start aiming, then tap a receiver with a second finger: the QB fires
# a fast, flat bullet at where he'll be (needs mod build v90 or later; on v89,
# throw normally and the receiver nearest the landing spot runs under it).
# No interception, breakup or tip on a super throw. Bullet speed, arrival height
# and how close the tap must be are on the Workshop's Tune tab (Super throw).
# ============================================================

on presnap
  # --- Super throw: change these ---
  super_on = 1                # 1 = on, 0 = off
  meter_per_catch = 20        # meter gained per completion (100 = full)
  meter_per_miss = 10         # meter lost per incompletion or interception
  super_needs = 100           # meter needed for a super throw
  super_clear_defenders = 1   # 1 = the 2 defenders nearest the catch are pushed away from it
  if meter == nil
    meter = 0
  end
end

on presnap
  if super_on
    msg("Super meter " + str(min(100, floor(meter * 100 / super_needs))) + "%")
    if meter >= super_needs
      button("Super throw")
    end
  end
end

on catch
  if super_on and not pressed("Super throw")
    meter = min(super_needs, meter + meter_per_catch)
    log("s1: meter", meter)
  end
end

on incomplete
  if super_on and not pressed("Super throw")
    meter = max(0, meter - meter_per_miss)
    log("s1: meter", meter)
  end
end

on intercepted
  if super_on and not pressed("Super throw")
    meter = max(0, meter - meter_per_miss)
    log("s1: meter", meter)
  end
end

on snap
  if super_on and pressed("Super throw")
    # no pick, no breakup, the defender always reads it late, lowest possible tip
    tune("pick_base", 0)
    tune("pick_known", 0)
    tune("breakup_share", 0)
    tune("whiff_tight", 1)
    tune("whiff_loose", 1)
    tune("tip_height", 15)
    log("s1: super throw")
  end
end

on frame
  if super_on and pressed("Super throw") and thrown() and landing_x() != nil
    for r in nearest(receivers(), landing_x(), landing_y(), 1)
      steer(r, landing_x(), landing_y(), 1)
    end
    # the defenders nearest the catch are pushed away from it while the ball is in the air
    if super_clear_defenders
      for d in nearest(defenders(), landing_x(), landing_y(), 2)
        steer(d, d.x + (d.x - landing_x()) * 2, d.y + (d.y - landing_y()) * 2, 1)
      end
    end
  end
end

on play_end
  if super_on and pressed("Super throw")
    meter = 0
  end
end

# ============================================================
# 2. SCREEN PASS
# Call one of the game's own plays where the back slips out, then tap "Screen".
# The back runs the play as normal. After a short pass-block, the linemen
# nearest him let their man go and run out ahead of him to block.
# ============================================================

on presnap
  # --- Screen pass: change these ---
  screen_on = 1          # 1 = on, 0 = off
  screen_delay = 0.8     # seconds the line pass-blocks before releasing
  screen_lead = 3        # how many yards in front of the back the blockers aim for
  screen_linemen = 3     # how many linemen release to block
  screen_time = 4        # seconds after the snap the blockers keep leading him
  if screen_on
    button("Screen")
  end
end

on snap
  scr_ready = 0
  if screen_on and pressed("Screen") and back() != nil
    scr_logged = 0
    scr_ready = 1
    log("s1: screen")
  end
end

on frame
  if scr_ready and back() != nil and since_snap() >= screen_delay and since_snap() < screen_time
    # aim a few yards in front of wherever the back is now, spread across his path
    tx = back().x + fwd() * yard(screen_lead)
    n = 0
    for p in nearest(offense(), back().x, back().y, screen_linemen + 4)
      ok = 1
      if p == qb() or p == back()
        ok = 0
      end
      for r in receivers()
        if p == r
          ok = 0
        end
      end
      if ok and n < screen_linemen
        steer(p, tx, back().y + (n - 1) * yard(1.5), 1)
        n = n + 1
        if not scr_logged
          log("s1: screen blocker", p.x, p.y)
        end
      end
    end
    scr_logged = 1
  end
end

# ============================================================
# 3. STOP-AND-GO
# Receivers make a sharp fake cut toward the middle, then break straight downfield.
# (The game's own slant-and-go route crashes the game, so the whole move is steered.)
# ============================================================

on presnap
  # --- Stop and go: change these ---
  sng_on = 1            # 1 = on, 0 = off
  sng_fake_time = 0.6   # seconds of the fake cut before they break downfield
  sng_fake_sharp = 1    # how sharp the fake cut is: 1 = 45 degrees, 2 = steeper, 0.5 = shallower
  if sng_on
    button("Stop and go")
  end
end

on snap
  if sng_on and pressed("Stop and go")
    log("s1: stop and go")
  end
end

on frame
  if sng_on and pressed("Stop and go") and not thrown()
    mid = (field_top() + field_bottom()) / 2
    for r in receivers()
      # with Screen also on, leave the back to the screen
      if not (scr_ready and r == back())
        if since_snap() < sng_fake_time
          dy = 1
          if r.y > mid
            dy = -1
          end
          steer(r, r.x + fwd() * yard(3), r.y + dy * yard(3) * sng_fake_sharp, 1)
        else
          steer(r, r.x + fwd() * yard(10), r.y, 1)
        end
      end
    end
  end
end

# ============================================================
# 4. LIGHT INJURIES
# After a tackle or sack, there's a small chance your QB or running back is
# hobbled: slower for the rest of the game. A message says so before the snap.
# shortcut: only the QB and the back - the game has no value that tells other
# players apart between plays; receivers and linemen need the step 2 build.
# ============================================================

on presnap
  # --- Injuries: change these ---
  injury_on = 1        # 1 = on, 0 = off
  injury_chance = 3    # percent chance per tackle or sack
  injury_speed = 85    # a hobbled player's speed, in percent
  injury_sacks = 1     # 1 = sacks can injure the QB, 0 = they can't
  no_tackle = 0
  if last_q != nil and quarter() < last_q
    hurt_qb = 0
    hurt_back = 0
    if injury_on
      log("s1: new game, injuries cleared")
    end
  end
  last_q = quarter()
  if injury_on and hurt_new
    msg("Injury! Your " + hurt_who + " is hobbled for the rest of the game")
  end
  hurt_new = 0
end

on incomplete
  no_tackle = 1
end

on touchdown
  no_tackle = 1
end

on intercepted
  no_tackle = 1
end

# "on play_end" doesn't run after a sack, so the QB gets his own roll here.
on sack
  if injury_on and injury_sacks and chance(injury_chance)
    hurt_qb = 1
    hurt_new = 1
    hurt_who = "QB"
    log("s1: QB injured on a sack")
  end
end

# A tackle: if the player brought down is the QB or the back, roll for an injury.
on play_end
  if injury_on and not no_tackle
    near = nearest(offense(), ball().x, ball().y, 1)
    if count(near) > 0 and chance(injury_chance)
      p = item(near, 0)
      if p == qb()
        hurt_qb = 1
        hurt_new = 1
        hurt_who = "QB"
        log("s1: QB injured on a tackle")
      elif p == back()
        hurt_back = 1
        hurt_new = 1
        hurt_who = "running back"
        log("s1: back injured on a tackle")
      end
    end
  end
end

on snap
  if injury_on and hurt_qb and qb() != nil
    speed(qb(), injury_speed)
    log("s1: hobbled QB")
  end
end

on snap
  if injury_on and hurt_back and back() != nil
    speed(back(), injury_speed)
    log("s1: hobbled back")
  end
end

# ============================================================
# 5. TRACE
# Records where every player and the ball are, about 6 times a second, so
# Claude can draw the play as a picture from rbmod.log.
# There's no Trace button (the game shows at most 3), so it switches itself on.
# ============================================================

on presnap
  # --- Trace: change these ---
  trace_mode = 1      # 0 = off, 1 = plays where you press Screen, Stop and go or Super throw, 2 = every play
  trace_every = 10    # frames between snapshots (60 frames = 1 second)
  tr_f = 0
end

on snap
  tr_on = 0
  if trace_mode == 2 or (trace_mode == 1 and (pressed("Screen") or pressed("Stop and go") or pressed("Super throw")))
    tr_on = 1
    log("s1: TS", los_x(), fwd(), field_top(), field_bottom(), down())
  end
end

# One snapshot: the ball, then each player as  time, role, number, x, y.
# Roles: 1 QB, 2 back, 3 receiver, 4 lineman, 5 defender.
on frame
  if tr_on
    tr_f = tr_f + 1
    if trace_every <= 1 or tr_f % trace_every == 1
      t = since_snap()
      log("s1: TB", t, ball().x, ball().y, state())
      i = 0
      for p in offense()
        k = 4
        if p == qb()
          k = 1
        elif p == back()
          k = 2
        else
          for r in receivers()
            if p == r
              k = 3
            end
          end
        end
        log("s1: TP", t, k, i, p.x, p.y)
        i = i + 1
      end
      i = 0
      for p in defenders()
        log("s1: TP", t, 5, i, p.x, p.y)
        i = i + 1
      end
    end
  end
end
