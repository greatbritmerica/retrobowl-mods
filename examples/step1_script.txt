# Step 1 play scripts: super bullet throw, screen pass, stop-and-go, light injuries.
# Needs mod build v89 or later. Paste into the Workshop's Script tab (below the
# Hail Mary script if you use it) and Save to Files as rbmod_script.txt.
# To tune a feature, change the numbers in its first "on presnap" block.
# To turn a feature off, set its  ..._on  line to 0.
# Everything this file does is logged in rbmod.log on lines starting with "s1:".

# ============================================================
# 1. SUPER BULLET THROW
# Completions fill a meter. When it's full, a "Super throw" button appears:
# tap it, throw, and the receiver nearest the landing spot runs under it
# with no interception chance.
# ============================================================

on presnap
  # --- Super throw: change these ---
  super_on = 1                # 1 = on, 0 = off
  meter_per_catch = 20        # meter gained per completion (100 = full)
  meter_per_miss = 10         # meter lost per incompletion or interception
  super_needs = 100           # meter needed for a super throw
  super_slow_defenders = 1    # 1 = the 2 defenders nearest the catch are slowed
  super_def_speed = 70        # their speed during a super throw, in percent
  if meter == nil
    meter = 0
  end
end

on presnap
  if super_on
    msg("Super meter " + str(min(100, floor(meter * 100 / super_needs))) + "%")
    if meter >= super_needs
      button("Super throw")
    end
  end
end

on catch
  if super_on and not pressed("Super throw")
    meter = min(super_needs, meter + meter_per_catch)
    log("s1: meter", meter)
  end
end

on incomplete
  if super_on and not pressed("Super throw")
    meter = max(0, meter - meter_per_miss)
    log("s1: meter", meter)
  end
end

on intercepted
  if super_on and not pressed("Super throw")
    meter = max(0, meter - meter_per_miss)
    log("s1: meter", meter)
  end
end

on snap
  if super_on and pressed("Super throw")
    tune("pick_base", 0)
    tune("pick_known", 0)
    log("s1: super throw")
  end
end

on frame
  if super_on and pressed("Super throw") and thrown()
    for r in nearest(receivers(), landing_x(), landing_y(), 1)
      steer(r, landing_x(), landing_y(), 1)
    end
    if super_slow_defenders
      for d in nearest(defenders(), landing_x(), landing_y(), 2)
        speed(d, super_def_speed)
      end
    end
  end
end

on play_end
  if super_on and pressed("Super throw")
    meter = 0
  end
end
