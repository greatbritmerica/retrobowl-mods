# Step 1 play scripts: super bullet throw, screen pass, stop-and-go, light injuries.
# Needs mod build v89 or later. Paste into the Workshop's Script tab (below the
# Hail Mary script if you use it) and Save to Files as rbmod_script.txt.
# To tune a feature, change the numbers in its first "on presnap" block.
# To turn a feature off, set its  ..._on  line to 0.
# Everything this file does is logged in rbmod.log on lines starting with "s1:".
# Play on EXTREME difficulty: the meter and injuries use events the mod only
# sends on Extreme. The screen pass and stop-and-go work on any difficulty.
# The game shows at most 3 buttons before a snap. When the Hail Mary button is
# up (end of a half), one of these may not appear.

# ============================================================
# 1. SUPER BULLET THROW
# Completions fill a meter. When it's full, a "Super throw" button appears:
# tap it, throw, and the receiver nearest the landing spot runs under it
# with no interception chance.
# ============================================================

on presnap
  # --- Super throw: change these ---
  super_on = 1                # 1 = on, 0 = off
  meter_per_catch = 20        # meter gained per completion (100 = full)
  meter_per_miss = 10         # meter lost per incompletion or interception
  super_needs = 100           # meter needed for a super throw
  super_slow_defenders = 1    # 1 = the 2 defenders nearest the catch are slowed
  super_def_speed = 70        # their speed during a super throw, in percent
  if meter == nil
    meter = 0
  end
end

on presnap
  if super_on
    msg("Super meter " + str(min(100, floor(meter * 100 / super_needs))) + "%")
    if meter >= super_needs
      button("Super throw")
    end
  end
end

on catch
  if super_on and not pressed("Super throw")
    meter = min(super_needs, meter + meter_per_catch)
    log("s1: meter", meter)
  end
end

on incomplete
  if super_on and not pressed("Super throw")
    meter = max(0, meter - meter_per_miss)
    log("s1: meter", meter)
  end
end

on intercepted
  if super_on and not pressed("Super throw")
    meter = max(0, meter - meter_per_miss)
    log("s1: meter", meter)
  end
end

on snap
  if super_on and pressed("Super throw")
    super_slowed = 0
    tune("pick_base", 0)
    tune("pick_known", 0)
    log("s1: super throw")
  end
end

on frame
  if super_on and pressed("Super throw") and thrown() and landing_x() != nil
    for r in nearest(receivers(), landing_x(), landing_y(), 1)
      steer(r, landing_x(), landing_y(), 1)
    end
    # speed() stacks each time it's called, so slow them once per throw
    if super_slow_defenders and not super_slowed
      for d in nearest(defenders(), landing_x(), landing_y(), 2)
        speed(d, super_def_speed)
      end
      super_slowed = 1
    end
  end
end

on play_end
  if super_on and pressed("Super throw")
    meter = 0
  end
end

# ============================================================
# 2. SCREEN PASS
# A "Screen" button. At the snap the back drifts out to one side behind the
# line, and the linemen nearest him release to set up just in front of him.
# ============================================================

on presnap
  # --- Screen pass: change these ---
  screen_on = 1          # 1 = on, 0 = off
  screen_side = 0        # 1 = top side, -1 = bottom side, 0 = the side with more room
  screen_depth = 5       # how many yards toward the sideline the back drifts
  screen_linemen = 3     # how many linemen release to block
  screen_time = 3        # seconds after the snap the blockers keep heading out
  if screen_on
    button("Screen")
  end
end

on snap
  scr_ready = 0
  if screen_on and pressed("Screen") and back() != nil
    scr_dy = 1
    if screen_side == 1
      scr_dy = -1
    elif screen_side == 0 and back().y - field_top() > field_bottom() - back().y
      scr_dy = -1
    end
    scr_y = max(field_top() + 20, min(field_bottom() - 20, back().y + scr_dy * yard(screen_depth)))
    # the back waits 2 yards behind the line; the blockers set up just past it, in front of him
    scr_bx = los_x() - fwd() * yard(2)
    scr_x = los_x() + fwd() * yard(1)
    scr_logged = 0
    scr_ready = 1
    log("s1: screen side", scr_dy, "spot", scr_x, scr_y)
  end
end

# Kept apart: route() fails if the back is already on that route, and that
# only switches off this block.
on snap
  if scr_ready
    route(back(), "route_flat")
  end
end

on frame
  if scr_ready and back() != nil and since_snap() < screen_time
    # the game's flat route can head either way, so steer the back to the blockers' side
    if not thrown()
      steer(back(), scr_bx, scr_y, 1)
    end
    n = 0
    for p in nearest(offense(), los_x(), back().y, screen_linemen + 3)
      ok = 1
      if p == qb() or p == back()
        ok = 0
      end
      for r in receivers()
        if p == r
          ok = 0
        end
      end
      if ok and n < screen_linemen
        steer(p, scr_x, scr_y, 1)
        n = n + 1
        if not scr_logged
          log("s1: screen blocker", p.x, p.y)
        end
      end
    end
    scr_logged = 1
  end
end

# ============================================================
# 3. STOP-AND-GO
# Receivers start on a slant, then break straight downfield.
# (The game's own slant-and-go route crashes the game, so this steers them.)
# ============================================================

on presnap
  # --- Stop and go: change these ---
  sng_on = 1            # 1 = on, 0 = off
  sng_fake_time = 0.9   # seconds of slant before they break downfield
  if sng_on
    button("Stop and go")
  end
end

on snap
  if sng_on and pressed("Stop and go")
    log("s1: stop and go")
  end
end

# Kept apart: route() fails if a receiver is already on a slant, and that
# only switches off this block; the break downfield below still happens.
on snap
  if sng_on and pressed("Stop and go")
    for r in receivers()
      # with Screen also on, leave the back to the screen
      if not (scr_ready and r == back())
        route(r, "route_slant")
      end
    end
  end
end

on frame
  if sng_on and pressed("Stop and go") and since_snap() >= sng_fake_time and not thrown()
    for r in receivers()
      if not (scr_ready and r == back())
        steer(r, r.x + fwd() * yard(10), r.y, 1)
      end
    end
  end
end

# ============================================================
# 4. LIGHT INJURIES
# After a tackle or sack, there's a small chance the player who went down is
# hobbled: slower for the rest of the game. A message says so before the snap.
# The "injured id" lines in the log tell us which game value recognizes the
# same player on later plays; until that's confirmed, the slowdown may not stick.
# ============================================================

on presnap
  # --- Injuries: change these ---
  injury_on = 1        # 1 = on, 0 = off
  injury_chance = 3    # percent chance per tackle or sack
  injury_speed = 85    # a hobbled player's speed, in percent
  injury_sacks = 1     # 1 = sacks can injure the QB, 0 = they can't
  no_tackle = 0
  if last_q != nil and quarter() < last_q
    hurt_id = nil
    hurt_qb = 0
    log("s1: new game, injuries cleared")
  end
  last_q = quarter()
  if injury_on and hurt_new
    msg("Injury! A player is hobbled for the rest of the game")
  end
  hurt_new = 0
end

on incomplete
  no_tackle = 1
end

on touchdown
  no_tackle = 1
end

# "on play_end" doesn't run after a sack, so the QB gets his own roll here.
# He's always qb(), so no id is needed to find him again.
on sack
  if injury_on and injury_sacks and chance(injury_chance)
    hurt_qb = 1
    hurt_new = 1
    log("s1: QB injured on a sack")
  end
end

on play_end
  if injury_on and not no_tackle
    near = nearest(offense(), ball().x, ball().y, 1)
    if count(near) > 0 and chance(injury_chance)
      p = item(near, 0)
      hurt_x = p.x
      hurt_y = p.y
      hurt_new = 1
      log("s1: injured at", p.x, p.y)
    end
  end
end

# One block per name, so a name the game doesn't have only switches off its own block.
on play_end
  if hurt_new
    p = item(nearest(offense(), hurt_x, hurt_y, 1), 0)
    hurt_id = get(p, "id")
    log("s1: injured id id", hurt_id)
  end
end

on play_end
  if hurt_new
    log("s1: injured id player_id", get(item(nearest(offense(), hurt_x, hurt_y, 1), 0), "player_id"))
  end
end

on play_end
  if hurt_new
    log("s1: injured id uid", get(item(nearest(offense(), hurt_x, hurt_y, 1), 0), "uid"))
  end
end

on play_end
  if hurt_new
    log("s1: injured id number", get(item(nearest(offense(), hurt_x, hurt_y, 1), 0), "number"))
  end
end

on play_end
  if hurt_new
    log("s1: injured id slot", get(item(nearest(offense(), hurt_x, hurt_y, 1), 0), "slot"))
  end
end

on snap
  if injury_on and hurt_id != nil
    for p in offense()
      if get(p, "id") == hurt_id
        speed(p, injury_speed)
        log("s1: hobbled", hurt_id)
      end
    end
  end
end

on snap
  if injury_on and hurt_qb and qb() != nil
    speed(qb(), injury_speed)
    log("s1: hobbled QB")
  end
end

# ============================================================
# 5. TRACE
# Records where every player and the ball are, about 6 times a second, so
# Claude can draw the play as a picture from rbmod.log.
# There's no Trace button (the game shows at most 3), so it switches itself on.
# ============================================================

on presnap
  # --- Trace: change these ---
  trace_mode = 1      # 0 = off, 1 = plays where you press Screen, Stop and go or Super throw, 2 = every play
  trace_every = 10    # frames between snapshots (60 frames = 1 second)
  tr_f = 0
end

on snap
  tr_on = 0
  if trace_mode == 2 or (trace_mode == 1 and (pressed("Screen") or pressed("Stop and go") or pressed("Super throw")))
    tr_on = 1
    log("s1: TS", los_x(), fwd(), field_top(), field_bottom(), down())
  end
end

# One snapshot: the ball, then each player as  time, role, number, x, y.
# Roles: 1 QB, 2 back, 3 receiver, 4 lineman, 5 defender.
on frame
  if tr_on
    tr_f = tr_f + 1
    if trace_every <= 1 or tr_f % trace_every == 1
      t = since_snap()
      log("s1: TB", t, ball().x, ball().y, state())
      i = 0
      for p in offense()
        k = 4
        if p == qb()
          k = 1
        elif p == back()
          k = 2
        else
          for r in receivers()
            if p == r
              k = 3
            end
          end
        end
        log("s1: TP", t, k, i, p.x, p.y)
        i = i + 1
      end
      i = 0
      for p in defenders()
        log("s1: TP", t, 5, i, p.x, p.y)
        i = i + 1
      end
    end
  end
end
