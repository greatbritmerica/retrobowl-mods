# Step 1 play scripts: super bullet throw, screen pass, stop-and-go, light injuries.
# Needs mod build v89 or later. Paste into the Workshop's Script tab (below the
# Hail Mary script if you use it) and Save to Files as rbmod_script.txt.
# To tune a feature, change the numbers in its first "on presnap" block.
# To turn a feature off, set its  ..._on  line to 0.
# Everything this file does is logged in rbmod.log on lines starting with "s1:".
# Play on EXTREME difficulty: the meter and injuries use events the mod only
# sends on Extreme. The screen pass and stop-and-go work on any difficulty.
# The game shows at most 3 buttons before a snap. When the Hail Mary button is
# up (end of a half), one of these may not appear.

# ============================================================
# 1. SUPER BULLET THROW
# Completions fill a meter. When it's full, a "Super throw" button appears.
# Tap it, start aiming, then tap a receiver with a second finger: the QB fires
# a fast, flat bullet at where he'll be (needs mod build v90 or later; on v89,
# throw normally and the receiver nearest the landing spot runs under it).
# No interception, breakup or tip on a super throw. Bullet speed, arrival height
# and how close the tap must be are on the Workshop's Tune tab (Super throw).
# ============================================================

on presnap
  # --- Super throw: change these ---
  super_on = 1                # 1 = on, 0 = off
  meter_per_catch = 20        # meter gained per completion (100 = full)
  meter_per_miss = 0          # meter lost per incompletion or interception (0 = the meter only goes up until used)
  super_needs = 100           # meter needed for a super throw
  super_clear_defenders = 1   # 1 = the 2 defenders nearest the catch are pushed away from it
  if meter == nil
    meter = 0
  end
end

on presnap
  if super_on
    msg("Super meter " + str(min(100, floor(meter * 100 / super_needs))) + "%")
    if meter >= super_needs
      button("Super throw")
    end
  end
end

on catch
  if super_on and not pressed("Super throw")
    meter = min(super_needs, meter + meter_per_catch)
    log("s1: meter", meter)
  end
end

on incomplete
  if super_on and not pressed("Super throw")
    meter = max(0, meter - meter_per_miss)
    log("s1: meter", meter)
  end
end

on intercepted
  if super_on and not pressed("Super throw")
    meter = max(0, meter - meter_per_miss)
    log("s1: meter", meter)
  end
end

on snap
  if super_on and pressed("Super throw")
    # no pick, no breakup, the defender always reads it late, lowest possible tip
    tune("pick_base", 0)
    tune("pick_known", 0)
    tune("breakup_share", 0)
    tune("whiff_tight", 1)
    tune("whiff_loose", 1)
    tune("tip_height", 15)
    log("s1: super throw")
  end
end

on frame
  if super_on and pressed("Super throw") and thrown() and landing_x() != nil
    for r in nearest(receivers(), landing_x(), landing_y(), 1)
      steer(r, landing_x(), landing_y(), 1)
    end
    # the defenders nearest the catch are pushed away from it while the ball is in the air
    if super_clear_defenders
      for d in nearest(defenders(), landing_x(), landing_y(), 2)
        steer(d, d.x + (d.x - landing_x()) * 2, d.y + (d.y - landing_y()) * 2, 1)
      end
    end
  end
end

on play_end
  if super_on and pressed("Super throw")
    meter = 0
  end
end

# ============================================================
# 2. SCREEN PASS
# Tap "Screen". The back chips for a moment, then slips out to the side with
# more room and sets up just behind the line while the line keeps blocking.
# Once he's nearly set, the linemen on that side let their man go and form a
# wall just past the line; after the catch they lead him upfield.
# ============================================================

on presnap
  # --- Screen pass: change these ---
  screen_on = 1          # 1 = on, 0 = off
  screen_slip = 0.3      # seconds the back blocks before he slips out
  screen_block = 1.2     # seconds the line blocks before releasing to form the wall
  screen_out = 5         # yards the back slips out toward the sideline
  screen_behind = 2      # yards behind the line of scrimmage he sets up
  screen_linemen = 3     # how many linemen release to form the wall
  screen_lead = 3        # after the catch, how many yards in front of him the blockers run
  screen_time = 5        # seconds after the snap the blockers keep helping
  if screen_on
    button("Screen")
  end
end

on snap
  scr_ready = 0
  if screen_on and pressed("Screen") and back() != nil
    scr_air = 0
    scr_done = 0
    scr_ready = 1
    # his own side of the QB (crossing behind the QB runs him through the rush);
    # lined up straight behind the QB: the side with more room
    scr_dy = 1
    if qb() != nil and back().y < qb().y - 5
      scr_dy = -1
    elif (qb() == nil or back().y <= qb().y + 5) and back().y - field_top() > field_bottom() - back().y
      scr_dy = -1
    end
    scr_bx = los_x() - fwd() * yard(screen_behind)
    scr_by = max(field_top() + 20, min(field_bottom() - 20, back().y + scr_dy * yard(screen_out)))
    log("s1: screen spot", scr_bx, scr_by, "out/wall at", screen_slip, screen_block)
  end
end

on frame
  if scr_ready and back() != nil and since_snap() < screen_time
    # the pass is over once a throw has been in the air and has come down
    if thrown()
      scr_air = 1
    elif scr_air
      scr_done = 1
    end
    if not scr_done
      if since_snap() < screen_slip
        # strength 0 = no push: he stands in and blocks
        steer(back(), back().x, back().y, 0)
      else
        # full push until he's within a yard of the spot, then ease off so he settles there
        # (also while the ball is in the air, or the game's route runs him away from it)
        steer(back(), scr_bx, scr_by, min(1, dist(back().x, back().y, scr_bx, scr_by) / yard(1)))
      end
    end
    if since_snap() >= screen_block
      # before the catch: a wall just past the line, in front of his spot
      tx = los_x() + fwd() * yard(1)
      ty = scr_by
      if scr_done
        # after it: lead him upfield, never back behind the line
        tx = back().x
        if (tx - los_x()) * fwd() < 0
          tx = los_x()
        end
        tx = tx + fwd() * yard(screen_lead)
        # stay in his lane: halfway between the screen spot and where he is now
        ty = (scr_by + back().y) / 2
      end
      n = 0
      for p in nearest(offense(), scr_bx, scr_by, screen_linemen + 4)
        ok = 1
        if p == qb() or p == back()
          ok = 0
        end
        for r in receivers()
          if p == r
            ok = 0
          end
        end
        if ok and n < screen_linemen
          steer(p, tx, ty + (n - 1) * yard(2) * scr_dy, 1)
          n = n + 1
        end
      end
    end
  end
end

# ============================================================
# 3. MINUTE DRILL (needs mod build v94)
# Tap "Minute drill" to go up-tempo for the rest of this drive. Snap within
# md_window seconds of the last play ending and that play is on tempo:
#  - the defense isn't set: every defender reacts slowly for the first moment
#  - a quick, short catch leaves the nearest defenders a step slow
#  - any catch past md_juke_yards gets a JUKE button when a tackler closes: tap it to shake him
#  - on a handoff, your receivers go block the nearest defenders
# A slow snap keeps the mode on but earns nothing that play.
# (The catch rewards use catch events, so they work on Extreme only.)
# ============================================================

on presnap
  # --- Minute drill: change these ---
  md_on = 1             # 1 = on, 0 = off
  md_window = 7         # seconds after a play ends you have to snap to stay on tempo
  md_quick = 1.5        # a throw out within this many seconds of the snap counts as quick
  md_short = 5          # a catch within this many yards past the line counts as short
  md_slow_n = 3         # how many defenders a quick short catch slows
  md_slow_pct = 45      # how fast slowed and juked defenders move, in percent
  md_juke_yards = 3     # catches longer than this get the JUKE button
  md_juke_push = 3      # yards a juked tackler is pushed off
  md_push_time = 0.4    # seconds the juked tackler is pushed
  md_block_pct = 55     # a run-blocked defender's speed, in percent (one shared by two receivers can end up slower)
  md_block_time = 4     # seconds after the snap your receivers keep blocking on a run
  md_daze_pct = 60      # on a tempo snap the defense isn't set: how fast every defender reacts at first, in percent
  md_daze_time = 1      # seconds after a tempo snap before the defense is back to normal
  if md_on
    if md_active
      if md_was_slow
        msg("Last snap too slow. Minute drill: snap fast!")
      else
        msg("Minute drill: snap fast!")
      end
    else
      button("Minute drill")
    end
  end
end

on snap
  md_ok = 0
  md_was_slow = 0
  md_throw_t = 99
  md_push_until = 0
  md_run = 0
  if md_on and not md_active and pressed("Minute drill")
    md_active = 1
    md_ok = 1
    log("s1: md on")
  elif md_on and md_active
    if tempo() >= 0 and tempo() <= md_window
      md_ok = 1
    else
      md_was_slow = 1
    end
  end
  md_dazed = 0
  if md_ok and md_daze_pct > 0
    # the defense isn't set yet: everyone reacts slowly for a moment
    for p in defenders()
      speed(p, md_daze_pct)
    end
    md_dazed = 1
  end
  if md_ok and state() == 19
    md_run = 1
    # each receiver's nearest defender is slowed once (skip a repeat of the last one)
    md_tk = nil
    for r in sort_y(receivers())
      if r != back()
        p = item(nearest(defenders(), r.x, r.y, 1), 0)
        if p != nil and p != md_tk
          speed(p, md_block_pct)
          md_tk = p
        end
      end
    end
  end
  if md_active
    log("s1: md snap tempo", tempo(), "on", md_ok, "run", md_run)
  end
end

on throw
  md_throw_t = since_snap()
end

on catch
  if md_ok
    tx = (ball().x - los_x()) * fwd() / yard(1)
    if md_throw_t <= md_quick and tx <= md_short
      for p in nearest(defenders(), ball().x, ball().y, md_slow_n)
        speed(p, md_slow_pct)
      end
      log("s1: md quick catch", tx)
    end
    if tx > md_juke_yards
      p = item(nearest(offense(), ball().x, ball().y, 1), 0)
      if p != nil
        juke_ready(p)
        log("s1: md juke armed", tx)
      end
    end
  end
end

on juke
  md_tk = tackler()
  if md_tk != nil
    md_push_until = since_snap() + md_push_time
    speed(md_tk, md_slow_pct)
    log("s1: md juke")
  end
end

on frame
  if md_push_until > 0 and since_snap() < md_push_until
    # push the juked tackler straight away from your ball carrier (players are found again every block)
    p = tackler()
    if p != nil
      ty = max(1, dist(p.x, p.y, ball().x, ball().y))
      steer(p, p.x + (p.x - ball().x) / ty * yard(md_juke_push), p.y + (p.y - ball().y) / ty * yard(md_juke_push), 1)
    end
  end
end

on frame
  if md_dazed and since_snap() >= md_daze_time
    # the defense has set: back to normal speed (undo the slowdown, keep any other)
    for p in defenders()
      speed(p, 10000 / md_daze_pct)
    end
    md_dazed = 0
  end
end

on frame
  if md_run and since_snap() < md_block_time
    for r in receivers()
      if r != back()
        p = item(nearest(defenders(), r.x, r.y, 1), 0)
        if p != nil
          steer(r, p.x, p.y, 1)
        end
      end
    end
  end
end

on drive_end
  if md_active
    log("s1: md off")
  end
  md_active = 0
  md_ok = 0
  md_was_slow = 0
end

# ============================================================
# 4. LIGHT INJURIES
# After a tackle or sack, there's a small chance your QB or running back is
# hobbled: slower for the rest of the game. A message says so before the snap.
# shortcut: only the QB and the back - the game has no value that tells other
# players apart between plays; receivers and linemen need the step 2 build.
# ============================================================

on presnap
  # --- Injuries: change these ---
  injury_on = 1        # 1 = on, 0 = off
  injury_chance = 3    # percent chance per tackle or sack
  injury_speed = 85    # a hobbled player's speed, in percent
  injury_sacks = 1     # 1 = sacks can injure the QB, 0 = they can't
  no_tackle = 0
  if last_q != nil and quarter() < last_q
    hurt_qb = 0
    hurt_back = 0
    if injury_on
      log("s1: new game, injuries cleared")
    end
  end
  last_q = quarter()
  if injury_on and hurt_new
    msg("Injury! Your " + hurt_who + " is hobbled for the rest of the game")
  end
  hurt_new = 0
end

on incomplete
  no_tackle = 1
end

on touchdown
  no_tackle = 1
end

on intercepted
  no_tackle = 1
end

# "on play_end" doesn't run after a sack, so the QB gets his own roll here.
on sack
  if injury_on and injury_sacks and chance(injury_chance)
    hurt_qb = 1
    hurt_new = 1
    hurt_who = "QB"
    log("s1: QB injured on a sack")
  end
end

# A tackle: if the player brought down is the QB or the back, roll for an injury.
on play_end
  if injury_on and not no_tackle
    near = nearest(offense(), ball().x, ball().y, 1)
    if count(near) > 0 and chance(injury_chance)
      p = item(near, 0)
      if p == qb()
        hurt_qb = 1
        hurt_new = 1
        hurt_who = "QB"
        log("s1: QB injured on a tackle")
      elif p == back()
        hurt_back = 1
        hurt_new = 1
        hurt_who = "running back"
        log("s1: back injured on a tackle")
      end
    end
  end
end

on snap
  if injury_on and hurt_qb and qb() != nil
    speed(qb(), injury_speed)
    log("s1: hobbled QB")
  end
end

on snap
  if injury_on and hurt_back and back() != nil
    speed(back(), injury_speed)
    log("s1: hobbled back")
  end
end

# ============================================================
# 5. TRACE
# Records where every player and the ball are, about 6 times a second, so
# Claude can draw the play as a picture from rbmod.log.
# There's no Trace button (the game shows at most 3), so it switches itself on.
# ============================================================

on presnap
  # --- Trace: change these ---
  trace_mode = 1      # 0 = off, 1 = plays where you press Screen, Minute drill or Super throw, 2 = every play
  trace_every = 10    # frames between snapshots (60 frames = 1 second)
  tr_f = 0
end

on snap
  tr_on = 0
  if trace_mode == 2 or (trace_mode == 1 and (pressed("Screen") or pressed("Minute drill") or pressed("Super throw")))
    tr_on = 1
    log("s1: TS", los_x(), fwd(), field_top(), field_bottom(), down())
  end
end

# One snapshot: the ball, then each player as  time, role, number, x, y.
# Roles: 1 QB, 2 back, 3 receiver, 4 lineman, 5 defender.
on frame
  if tr_on
    tr_f = tr_f + 1
    if trace_every <= 1 or tr_f % trace_every == 1
      t = since_snap()
      log("s1: TB", t, ball().x, ball().y, state())
      i = 0
      for p in offense()
        k = 4
        if p == qb()
          k = 1
        elif p == back()
          k = 2
        else
          for r in receivers()
            if p == r
              k = 3
            end
          end
        end
        log("s1: TP", t, k, i, p.x, p.y)
        i = i + 1
      end
      i = 0
      for p in defenders()
        log("s1: TP", t, 5, i, p.x, p.y)
        i = i + 1
      end
    end
  end
end
